1. Setup & first test

Install pytest (preferably inside a virtual environment):

pip install pytest

Create test_sample.py. By default pytest discovers files named test_*.py or *_test.py, classes prefixed Test, and functions prefixed test_.

# test_sample.py
def test_always_passes():
    assert True

def test_arithmetic():
    assert 2 + 2 == 4
    assert 3 * 4 == 12

Run it:

pytest test_sample.py -v
test_sample.py::test_always_passes PASSED
test_sample.py::test_arithmetic PASSED
============================== 2 passed in 0.02s ===============================
💡 Use pytest -v (verbose) / -q (quiet). Running plain pytest discovers everything below the current directory.

2. Assertions & expected exceptions

pytest uses Python's built-in assert but adds rich introspection. Failed assertions show a diff of actual vs. expected values.

def test_failure_demo():
    expected = {"name": "Alice", "age": 30}
    actual = {"name": "Alice", "age": 31}
    assert actual == expected  # pytest shows a helpful diff

Testing exceptions with pytest.raises:

import pytest

def divide(a, b):
    if b == 0:
        raise ValueError("Cannot divide by zero")
    return a / b

def test_divide_by_zero():
    with pytest.raises(ValueError) as excinfo:
        divide(10, 0)
    assert "Cannot divide by zero" in str(excinfo.value)

def test_divide_zero_match():
    with pytest.raises(ValueError, match="divide by zero"):  # regex
        divide(1, 0)

Exception groups (Python 3.11+)

import pytest

def raises_group():
    raise ExceptionGroup("many errors", [ValueError("a"), TypeError("b")])

def test_group():
    with pytest.raises(ExceptionGroup) as exc:
        raises_group()
    assert len(exc.value.exceptions) == 2

3. Assertion rewriting — the magic explained advanced

pytest rewrites the AST of every test module before execution. That's why:

  • You write plain assert (no self.assertEqual).
  • Failed assertions show the values of intermediate expressions.
  • Asserts work even when Python is run with -O (which normally strips them).
def test_rewrite_demo():
    a, b = 1, 2
    assert a + b == 4
    # E   assert (1 + 2) == 4
    # E    +  where 1 = a
    # E    +  and   2 = b

Disable with --assert=plain (rarely useful) or per-module with PYTEST_DONT_REWRITE.

📌 Assertion rewriting does not work inside exec(), eval(), or list/generator comprehensions on some Python versions. Keep assertions simple.

4. Fixtures — the heart of pytest

Fixtures provide a baseline for tests: data, connections, temp files, or anything reusable. Defined with @pytest.fixture, injected by name.

import pytest

@pytest.fixture
def sample_list():
    return [1, 2, 3, 4, 5]

def test_sum(sample_list):
    assert sum(sample_list) == 15

def test_length(sample_list):
    assert len(sample_list) == 5

Fixture scopes

Default scope is function. Other scopes: class, module, package, session.

@pytest.fixture(scope="module")
def expensive_resource():
    print("\n[setup]")
    resource = {"data": [1, 2, 3]}
    yield resource
    print("\n[teardown]")

def test_one(expensive_resource):
    assert expensive_resource["data"][0] == 1

def test_two(expensive_resource):
    assert len(expensive_resource["data"]) == 3

Fixtures requesting other fixtures

@pytest.fixture
def base_url():
    return "https://api.example.com"

@pytest.fixture
def api_client(base_url):
    return {"base": base_url, "timeout": 5}

def test_client(api_client):
    assert api_client["base"].startswith("https")
🔁 Autouse: @pytest.fixture(autouse=True) applies a fixture to every test in scope without explicit injection. Use --setup-show to see the order.

5. Parametrization — one test, many cases

import pytest

def is_even(n):
    return n % 2 == 0

@pytest.mark.parametrize("number, expected", [
    (2, True), (3, False), (0, True), (-4, True), (7, False),
])
def test_is_even(number, expected):
    assert is_even(number) == expected

Stack decorators for combinations (4 runs here):

@pytest.mark.parametrize("x", [0, 1])
@pytest.mark.parametrize("y", [2, 3])
def test_combinations(x, y):
    assert x + y > 0

Use pytest.param for custom IDs, marks, or skip:

@pytest.mark.parametrize("value, expected", [
    pytest.param(2, 4, id="positive"),
    pytest.param(-2, 4, id="negative"),
    pytest.param(0, 0, id="zero"),
    pytest.param(5, 25, id="skip-me", marks=pytest.mark.skip(reason="example")),
])
def test_square(value, expected):
    assert value * value == expected

Readable IDs via ids

@pytest.fixture(params=[0, 1], ids=["zero", "one"])
def fixture_num(request):
    return request.param

# or on parametrize directly
@pytest.mark.parametrize("n", [0, 1], ids=["zero", "one"])
def test_ids(n):
    assert n in (0, 1)

Dynamic parametrization via pytest_generate_tests

# conftest.py
def pytest_generate_tests(metafunc):
    if "value" in metafunc.fixturenames:
        metafunc.parametrize("value", [10, 20, 30])

6. Marks: skip, xfail, and custom

  • @pytest.mark.skip — always skip.
  • @pytest.mark.skipif — skip conditionally.
  • @pytest.mark.xfail — expected to fail.
  • @pytest.mark.<name> — custom marks selectable with -m.
import sys
import pytest

@pytest.mark.skip(reason="feature not implemented yet")
def test_future(): ...

@pytest.mark.skipif(sys.version_info < (3, 10), reason="3.10+")
def test_new_syntax():
    assert (x := 5) == 5

@pytest.mark.xfail(reason="known issue #123")
def test_flaky():
    assert 1 == 2

@pytest.mark.slow
def test_slow_thing():
    import time; time.sleep(2)

Run only marked: pytest -m slow. Exclude: pytest -m "not slow". Register markers to silence warnings:

[pytest]
markers =
    slow: marks tests as slow
    integration: integration tests

7. Monkeypatch & mocking

monkeypatch temporarily modifies attributes, dicts, env vars, or sys.path, then restores automatically.

import os

def get_api_key():
    return os.environ.get("API_KEY", "default")

def test_api_key(monkeypatch):
    monkeypatch.setenv("API_KEY", "secret123")
    assert get_api_key() == "secret123"

def test_chdir(monkeypatch, tmp_path):
    monkeypatch.chdir(tmp_path)
    assert tmp_path == __import__("pathlib").Path.cwd()

def test_syspath(monkeypatch, tmp_path):
    monkeypatch.syspath_prepend(str(tmp_path))
    assert str(tmp_path) in __import__("sys").path

Replace attributes with monkeypatch.setattr:

import requests

def fetch_data(url):
    return requests.get(url).json()

def test_fetch_data(monkeypatch):
    class MockResponse:
        @staticmethod
        def json(): return {"mocked": True}
    def mock_get(*a, **k): return MockResponse()
    monkeypatch.setattr(requests, "get", mock_get)
    assert fetch_data("https://fake.com") == {"mocked": True}

For deeper control use unittest.mock (built-in) or the pytest-mock plugin:

from unittest.mock import patch

def test_with_patch():
    with patch("requests.get") as mock_get:
        mock_get.return_value.json.return_value = {"ok": True}
        assert fetch_data("http://test") == {"ok": True}
        mock_get.assert_called_once_with("http://test")

8. Built-in fixtures: tmp_path, capsys, caplog

tmp_path — temporary directories

def test_write_file(tmp_path):
    file = tmp_path / "data.txt"
    file.write_text("hello")
    assert file.read_text() == "hello"

def test_multiple_files(tmp_path):
    (tmp_path / "a.txt").write_text("A")
    (tmp_path / "b.txt").write_text("B")
    assert sorted(p.name for p in tmp_path.iterdir()) == ["a.txt", "b.txt"]

tmp_path_factory for session-scoped dirs. Legacy tmpdir (py.path) still exists — prefer tmp_path.

capsys — capture stdout / stderr

import sys

def test_print_output(capsys):
    print("hello world")
    print("this is stderr", file=sys.stderr)
    captured = capsys.readouterr()
    assert captured.out == "hello world\n"
    assert "stderr" in captured.err

Use capfd for file-descriptor-level output (C extensions).

caplog — capture logging

import logging

def test_logging(caplog):
    logger = logging.getLogger("myapp")
    with caplog.at_level(logging.WARNING):
        logger.warning("something bad")
        logger.info("ignored")
    assert "something bad" in caplog.text
    assert caplog.records[0].levelname == "WARNING"
    assert caplog.records[0].name == "myapp"

9. pytest.approx & warnings

Floating-point comparisons with tolerance:

import pytest

def test_float_equality():
    assert 0.1 + 0.2 == pytest.approx(0.3)
    assert 0.1 + 0.2 == pytest.approx(0.3, abs=1e-9)
    assert [0.1 + 0.2, 1.0] == pytest.approx([0.3, 1.0])
    assert {"x": 0.1 + 0.2} == pytest.approx({"x": 0.3})
    assert float("nan") == pytest.approx(float("nan"), nan_ok=True)

Testing warnings:

import warnings
import pytest

def old_function():
    warnings.warn("this is deprecated", DeprecationWarning)

def test_warning():
    with pytest.warns(DeprecationWarning, match="deprecated"):
        old_function()

def test_deprecated():
    with pytest.deprecated_call():
        old_function()

To ensure no warning is raised:

def test_no_warning():
    with warnings.catch_warnings():
        warnings.simplefilter("error")
        normal_function()

12. Import modes & the __init__.py question advanced

pytest has three import modes via --import-mode:

  • prepend (default) — inserts the test's directory at the front of sys.path. Simplest, but can shadow same-named modules across directories.
  • append — inserts at the end. Only relevant if you need a specific resolution order.
  • importlib — imports test modules by file path without touching sys.path. Requires __init__.py in test dirs to avoid module-name clashes.

Do you need __init__.py in tests?

  • Traditional project (tests import from an installed package): no.
  • Same test module name in two folders: yes, or use --import-mode=importlib.
  • Ship tests as a package: yes.
📌 Default behaviour works for 90% of projects. Only reach for --import-mode=importlib when you hit "import file mismatch" errors.

13. conftest.py hierarchy & fixture override rules advanced

conftest.py files are auto-discovered per directory. Fixtures defined there are available to all tests below that directory — no import needed.

my_project/
├── conftest.py # session-wide fixtures
├── tests/
│ ├── conftest.py # applies to tests/ and below
│ ├── test_api.py
│ └── integration/
│ ├── conftest.py # only this subfolder
│ └── test_db.py

Override rules: a fixture with the same name in a deeper conftest.py overrides the outer one. Tests can also override per-module by defining a local fixture.

# tests/conftest.py
import pytest

@pytest.fixture
def api_client():
    return {"env": "test"}

@pytest.fixture
def db():
    yield "real-db"
# tests/integration/conftest.py — overrides for this folder
import pytest

@pytest.fixture
def api_client():
    return {"env": "integration"}

@pytest.fixture
def db():
    yield "fake-db"
# tests/test_api.py
def test_env(api_client):
    assert api_client["env"] == "test"
# tests/integration/test_db.py
def test_env(api_client):
    assert api_client["env"] == "integration"
⚠️ conftest.py is not a plugin — it can't be imported by arbitrary modules and isn't distributed with your package. For reusable fixtures, ship a plugin.

14. Advanced fixtures & the request object

Fixtures can accept the special request fixture, exposing request.param, request.cls, request.node, request.config, and request.addfinalizer.

Parametrizing fixtures

@pytest.fixture(params=[1, 2, 3])
def number(request):
    return request.param

def test_number_positive(number):
    assert number > 0   # runs 3 times

Indirect parametrization

@pytest.fixture
def user(request):
    return {"id": request.param, "name": f"User{request.param}"}

@pytest.mark.parametrize("user", [1, 2], indirect=True)
def test_user_id(user):
    assert user["id"] in [1, 2]

Factory pattern (return a function from a fixture)

@pytest.fixture
def make_user():
    def _make_user(name, age):
        return {"name": name, "age": age}
    return _make_user

def test_multiple_users(make_user):
    alice = make_user("Alice", 30)
    bob = make_user("Bob", 25)
    assert alice["name"] != bob["name"]

Finalization with addfinalizer

@pytest.fixture
def resource(request):
    r = {"status": "open"}
    def teardown():
        r["status"] = "closed"
    request.addfinalizer(teardown)
    return r

def test_resource(resource):
    assert resource["status"] == "open"
📌 yield vs. addfinalizer: yield is cleaner for simple teardown. Use addfinalizer when you need conditional or multiple cleanup hooks.

15. Class-based tests

Group related tests in a class. setup_method/teardown_method run before/after each method; class-level variants are setup_class/teardown_class.

import pytest

class Calculator:
    def add(self, a, b): return a + b
    def divide(self, a, b):
        if b == 0: raise ValueError("division by zero")
        return a / b

class TestCalculator:
    def setup_method(self):
        self.calc = Calculator()

    def test_add(self):
        assert self.calc.add(1, 2) == 3

    def test_divide(self):
        assert self.calc.divide(10, 2) == 5.0

    def test_divide_zero(self):
        with pytest.raises(ValueError):
            self.calc.divide(1, 0)
💡 Prefer fixtures over setup_method — more composable. Class-based is useful for porting from unittest.

pytest runs unittest.TestCase subclasses too, so migration is gradual:

import unittest

class TestLegacy(unittest.TestCase):
    def test_equal(self):
        self.assertEqual(2 + 2, 4)

19. Running subsets — node ID syntax new

Every test has a node ID. You can address any subset precisely:

# single function
pytest tests/test_calc.py::test_add

# class + method
pytest tests/test_calc.py::TestCalc::test_add

# specific parametrized case
pytest tests/test_calc.py::test_add[1-2]

# keyword expressions
pytest -k "add and not slow"
pytest -k "TestCalc and divide"

# by marker
pytest -m "integration and not slow"

# last failed / failed first
pytest --lf
pytest --ff

# collect only (don't run)
pytest --collect-only
pytest --co -q     # quiet collection

20. Runtime helpers: skip, fail, xfail

import pytest

def test_conditional_skip():
    if not has_network():
        pytest.skip("no network available")
    assert True

def test_expected_failure():
    if not feature_flag_enabled():
        pytest.xfail("feature flag off")

def test_explicit_fail():
    if unexpected_condition():
        pytest.fail("something went wrong", pytrace=False)

# skip a whole module at import time
pytestmark = pytest.mark.skipif(no_gpu(), reason="requires GPU")

# fail based on environment
if not os.environ.get("RUN_SLOW"):
    pytestmark = pytest.mark.skip(reason="RUN_SLOW not set")

Use pytest.importorskip for optional dependencies:

def test_numpy():
    np = pytest.importorskip("numpy")
    assert np.array([1, 2]).sum() == 3

22. Hooks & writing your own plugin advanced

Hooks go in conftest.py or your own plugin module. The most-used ones:

# conftest.py
import pytest

def pytest_addoption(parser):
    parser.addoption("--env", action="store", default="dev",
                     help="target environment")

@pytest.fixture
def env(request):
    return request.config.getoption("--env")

def pytest_configure(config):
    config.addinivalue_line("markers", "smoke: quick smoke tests")

def pytest_collection_modifyitems(config, items):
    for item in items:
        if "integration" in item.nodeid:
            item.add_marker(pytest.mark.integration)

def pytest_report_teststatus(report):
    # custom status character
    if report.when == "call" and report.passed:
        return report.outcome, "P", "PASSED"

def pytest_sessionfinish(session, exitstatus):
    print(f"\nsession finished with status {exitstatus}")

CLI usage:

pytest --env=prod

Writing a distributable plugin

# myplugin.py
def pytest_configure(config):
    config.addinivalue_line("markers", "custom: my marker")
# pyproject.toml
[project.entry-points.pytest11]
myplugin = "myplugin"

After pip install -e ., your plugin loads automatically. Verify with pytest --trace-config.

23. Property-based testing with Hypothesis advanced

pip install hypothesis
from hypothesis import given, strategies as st

@given(st.integers(), st.integers())
def test_add_commutative(a, b):
    assert a + b == b + a

@given(st.lists(st.integers()))
def test_sorted_is_ordered(xs):
    assert all(sorted(xs)[i] <= sorted(xs)[i+1] for i in range(len(xs)-1))

Hypothesis generates hundreds of inputs (including edge cases) and shrinks failures to the smallest reproducer. Combine with pytest's normal fixtures.

25. Anti-patterns & pitfalls

❌ Test interdependence

# BAD — test_b assumes test_a ran
counter = 0
def test_a(): global counter; counter += 1
def test_b(): assert counter == 1

# GOOD — each test is self-contained
def increment(n): return n + 1

def test_a(): assert increment(0) == 1
def test_b(): assert increment(0) == 1

❌ Module-scoped mutable state leaking

# BAD — mutations leak across tests
@pytest.fixture(scope="module")
def shared_list():
    return []

# GOOD — copy, or use function scope
@pytest.fixture(scope="module")
def shared_list():
    data = []
    yield data
    data.clear()

❌ Over-mocking

If you mock everything, you're testing your mocks. Mock only at boundaries (network, DB, filesystem). Prefer fakes for simple cases.

❌ Slow fixtures without broader scope

A 500ms DB connection recreated per test adds up fast. Use scope="module" or "session" if tests don't mutate shared state.

❌ Many unrelated asserts in one test

The first failure hides the rest. Prefer one logical assertion per test, or use parametrize.

❌ Relying on print()

Use capsys / caplog so tests can assert on output.

❌ Assuming assert order doesn't matter

pytest's rewriting shows you values — but only for the first failed assert in a statement chain. Keep each statement focused.

26. Common errors → fixes new

ModuleNotFoundError: No module named 'myapp'

Test file can't find your package. Fix with one of:

  • pip install -e . (recommended)
  • Add pythonpath = ["src"] to pyproject.toml under [tool.pytest.ini_options]
  • Add a conftest.py at the repo root (pytest inserts rootdir on sys.path)
import file mismatch: imported module 'test_x' has this __file__ attribute...

Two test modules share a name across folders. Fix:

  • Add __init__.py to test folders, or
  • Run pytest --import-mode=importlib
PytestUnknownMarkWarning: Unknown pytest.mark.slow

Register custom marks in config:

[pytest]
markers =
    slow: marks tests as slow

Or run with --strict-markers to make it an error.

fixture 'xyz' not found

Check: spelling, that the fixture is defined in this file / a conftest.py above, and that @pytest.fixture decorator is present.

ScopeMismatch: You tried to access a 'function' scoped fixture from a 'module' scoped fixture

A wider-scope fixture requested a narrower one. Fix: broaden the inner fixture's scope, or restructure so the narrower fixture requests the wider one.

RuntimeError: no running event loop

Async test without @pytest.mark.asyncio (or asyncio_mode = "auto"). See §16.

DeprecationWarning: pytest.warns(None) is deprecated

Use warnings.catch_warnings() with warnings.simplefilter("error") instead. See §9.

Tests pass individually but fail together

State leaks between tests. Run pytest -p no:randomly --lf, or install pytest-randomly to catch ordering issues. Look for module-level globals, shared fixtures with scope="module", or mutated default arguments.

        with:
          python-version: ${{ matrix.python-version }}
          cache: pip

      - name: Install dependencies
        run: |
          python -m pip install --upgrade pip
          pip install -e ".[test]"

      - name: Run tests with coverage
        run: pytest --cov=myapp --cov-report=xml --cov-report=term-missing

      - name: Upload coverage to Codecov
        if: matrix.python-version == '3.12'
        uses: codecov/codecov-action@v4
        with:
          files: coverage.xml

# src/myapp/calculator.py
class Calculator:
    def add(self, a, b):
        return a + b
    def divide(self, a, b):
        if b == 0:
            raise ValueError("Division by zero")
        return a / b
    def is_positive(self, n):
        return n > 0
# tests/conftest.py
import pytest
from myapp.calculator import Calculator

@pytest.fixture
def calc():
    return Calculator()

@pytest.fixture
def fixture_file(tmp_path):
    f = tmp_path / "input.txt"
    f.write_text("1\n2\n3\n")
    return f
# tests/test_calculator.py
import pytest

@pytest.mark.parametrize("a, b, expected", [
    (2, 3, 5),
    (-1, 1, 0),
    (0, 0, 0),
    (100, -50, 50),
])
def test_add(calc, a, b, expected):
    assert calc.add(a, b) == expected

def test_divide(calc):
    assert calc.divide(10, 2) == 5.0

def test_divide_by_zero(calc):
    with pytest.raises(ValueError, match="Division by zero"):
        calc.divide(5, 0)

@pytest.mark.parametrize("n, expected", [
    (5, True), (-3, False), (0, False), (0.001, True)
])
def test_is_positive(calc, n, expected):
    assert calc.is_positive(n) == expected

def test_read_input_file(fixture_file):
    numbers = [int(line) for line in fixture_file.read_text().splitlines()]
    assert numbers == [1, 2, 3]
# tests/test_cli.py
def test_cli_prints_result(capsys, calc):
    print(calc.add(2, 3))
    captured = capsys.readouterr()
    assert captured.out == "5\n"
